#tuplas
personas = ("Ana", "Luis", "Carlos")

print("Tupla original:")
print(personas)

lista = list(personas)
lista.append("Marta")
personas = tuple(lista)

print("\nDespués de insertar:")
print(personas)

lista = list(personas)
lista[1] = "Pedro"
personas = tuple(lista)

print("\nDespués de modificar:")
print(personas)

lista = list(personas)
lista.remove("Carlos")
personas = tuple(lista)

print("\nDespués de eliminar:")
print(personas)
