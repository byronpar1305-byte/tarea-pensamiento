filas = int(input("INGRESE TAMAÑO DE FILA "))
columnas = int(input("INGRESE TAMAÑO DE COLUMNA "))

matriz1 = []
matriz2 = []

print("Primera matriz:")
for i in range(filas):
    fila = []
    for j in range(columnas):
        num = int(input(f"Valor [{i}][{j}]: "))
        fila.append(num)
    matriz1.append(fila)

print("Segunda matriz:")

for i in range(filas):
    fila = []
    for j in range(columnas):
        num = int(input(f"Valor [{i}][{j}]: "))
        fila.append(num)
    matriz2.append(fila)

resultado = []

for i in range(filas):
    fila = []
    for j in range(columnas):
        resta = matriz1[i][j] - matriz2[i][j]
        fila.append(resta)
    resultado.append(fila)

print("Resultado:")

for fila in resultado:
    print(fila)
