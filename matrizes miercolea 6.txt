matriz = []

print("Ingrese los valores de la matriz 3x3:")
for i in range(3):
    fila = []
    for j in range(3):
        num = int(input(f"Elemento [{i}][{j}]: "))
        fila.append(num)
    matriz.append(fila)

suma_filas = []
for fila in matriz:
    suma_filas.append(sum(fila))

suma_columnas = []
for j in range(3):
    suma = 0
    for i in range(3):
        suma += matriz[i][j]
    suma_columnas.append(suma)

print("\nMatriz:")
for fila in matriz:
    print(fila)

print("\nSuma de filas:", suma_filas)
print("Suma de columnas:", suma_columnas)

matriz = []
for i in range(3):
    fila = list(map(int, input("Ingrese 3 números separados por espacio: ").split()))
    matriz.append(fila)



filas = [sum(fila) for fila in matriz]

columnas = []
for j in range(3):
    columnas.append(matriz[0][j] + matriz[1][j] + matriz[2][j])


diag1 = matriz[0][0] + matriz[1][1] + matriz[2][2]
diag2 = matriz[0][2] + matriz[1][1] + matriz[2][0]

print("Suma filas:", filas)
print("Suma columnas:", columnas)
print("Diagonal 1:", diag1)
print("Diagonal 2:", diag2)


if filas[0] == filas[1] == filas[2] == columnas[0] == columnas[1] == columnas[2] == diag1 == diag2:
    print("Es un cuadrado mágico")
else:
    print("No es un cuadrado mágico")
